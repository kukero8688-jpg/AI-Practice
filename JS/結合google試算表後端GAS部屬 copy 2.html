若想讓刪除操作直接同步刪除 Google Sheet 試算表中的該行資料，只需將以下代碼覆蓋貼上至您的 Apps Script 專案並「重新部署」即可： 

function doGet(e) {
  try {
    var action = e && e.parameter ? String(e.parameter.action || '').trim() : '';

    if (action === 'delete') {
      var idFromGet = String(e.parameter.id || '').trim();
      var getLock = LockService.getScriptLock();
      getLock.waitLock(10000);
      try {
        return deleteMember_(idFromGet);
      } finally {
        try { getLock.releaseLock(); } catch (releaseErr) {}
      }
    }

    if (action !== 'read') {
      return jsonResponse_({
        status: 'error',
        message: '不支援的 GET action'
      });
    }

    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    return jsonResponse_({
      status: 'success',
      data: readMembers_(sheet)
    });

  } catch (err) {
    return jsonResponse_({
      status: 'error',
      message: String(err && err.message ? err.message : err)
    });
  }
}

function doPost(e) {
  var lock = LockService.getScriptLock();

  try {
    lock.waitLock(10000);

    var data = parsePostData_(e);
    var action = String(data.action || '').trim();

    // 刪除會員：只需要 action + id；不要求 email/name，避免誤擋刪除。
    if (action === 'delete') {
      var idToDelete = String(data.id ?? '').trim();
      return deleteMember_(idToDelete);
    }

    // 既有新增會員功能
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var name = String(data.name || '').trim();
    var email = String(data.email || '').trim();
    var role = String(data.role || '').trim();

    if (!name || !email || !role) {
      return jsonResponse_({
        status: 'error',
        message: '會員姓名、Email、角色皆不可為空白。'
      });
    }

    var nextId = getNextMemberId_(sheet);

    sheet.appendRow([
      nextId,
      name,
      email,
      role,
      new Date()
    ]);

    SpreadsheetApp.flush();

    return jsonResponse_({
      status: 'success',
      message: '資料新增成功',
      id: nextId
    });

  } catch (err) {
    return jsonResponse_({
      status: 'error',
      message: String(err && err.message ? err.message : err)
    });
  } finally {
    try {
      lock.releaseLock();
    } catch (lockErr) {
      // 忽略鎖定釋放錯誤
    }
  }
}

function deleteMember_(idToDelete) {
  var id = String(idToDelete || '').trim();

  if (!id) {
    return jsonResponse_({
      status: 'error',
      message: '缺少會員 ID，無法執行刪除。'
    });
  }

  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rowNumber = findMemberRowById_(sheet, id);

  if (rowNumber === -1) {
    return jsonResponse_({
      status: 'error',
      message: '找不到欲刪除的會員 ID #' + id
    });
  }

  sheet.deleteRow(rowNumber);
  SpreadsheetApp.flush();

  var verifyRow = findMemberRowById_(sheet, id);

  if (verifyRow !== -1) {
    return jsonResponse_({
      status: 'error',
      message: '刪除後驗證失敗：會員 #' + id + ' 仍存在。'
    });
  }

  return jsonResponse_({
    status: 'success',
    message: '會員 ID #' + id + ' 已確實從 Google Sheet 刪除。',
    deletedId: id
  });
}

function readMembers_(sheet) {
  var lastRow = sheet.getLastRow();
  var lastColumn = sheet.getLastColumn();

  if (lastRow < 2 || lastColumn < 1) {
    return [];
  }

  var values = sheet.getRange(1, 1, lastRow, Math.max(lastColumn, 4)).getValues();
  var members = [];

  for (var i = 1; i < values.length; i++) {
    var row = values[i];
    var id = String(row[0] ?? '').trim();
    var name = String(row[1] ?? '').trim();
    var email = String(row[2] ?? '').trim();
    var role = String(row[3] ?? '').trim();

    if (!id && !name && !email && !role) continue;

    members.push({
      id: id,
      name: name,
      email: email,
      role: role || '一般會員'
    });
  }

  return members;
}

function findMemberRowById_(sheet, idToFind) {
  var lastRow = sheet.getLastRow();

  if (lastRow < 2) return -1;

  var idValues = sheet.getRange(2, 1, lastRow - 1, 1).getDisplayValues();
  var target = String(idToFind).trim();

  for (var i = 0; i < idValues.length; i++) {
    if (String(idValues[i][0]).trim() === target) {
      return i + 2;
    }
  }

  return -1;
}

function getNextMemberId_(sheet) {
  var lastRow = sheet.getLastRow();

  if (lastRow < 2) return 1;

  var idValues = sheet.getRange(2, 1, lastRow - 1, 1).getValues();
  var maxId = 0;

  for (var i = 0; i < idValues.length; i++) {
    var value = Number(idValues[i][0]);
    if (isFinite(value) && value > maxId) maxId = value;
  }

  return maxId + 1;
}

function parsePostData_(e) {
  if (!e || !e.postData || !e.postData.contents) {
    throw new Error('沒有收到 POST 資料。');
  }

  var raw = String(e.postData.contents || '').trim();
  if (!raw) throw new Error('POST 資料為空。');

  return JSON.parse(raw);
}

function jsonResponse_(payload) {
  return ContentService
    .createTextOutput(JSON.stringify(payload))
    .setMimeType(ContentService.MimeType.JSON);
}
